    private int dp(float value) {
        return (int) (
                value * getResources().getDisplayMetrics().density
                + 0.5f
        );
    }

    private boolean shouldShowTvLogo() {
        Configuration c = getResources().getConfiguration();

        boolean landscape =
                c.orientation == Configuration.ORIENTATION_LANDSCAPE;

        boolean television =
                (c.uiMode & Configuration.UI_MODE_TYPE_MASK)
                == Configuration.UI_MODE_TYPE_TELEVISION;

        boolean tablet = c.smallestScreenWidthDp >= 600;

        return landscape && (television || tablet);
    }


    private int broadcastMiniWidth() {
        float density = getResources().getDisplayMetrics().density;
        int screenWidthDp = (int) (
                getResources().getDisplayMetrics().widthPixels / density
        );
        return dp(Math.max(220, Math.min(300, screenWidthDp - 24)));
    }

    private int broadcastMiniHeight() {
        return Math.round(broadcastMiniWidth() * 9f / 16f);
    }

    private void updateMiniLogoLayout(
            ImageView logo, boolean fullscreen) {

        if (logo == null) return;

        boolean portrait =
                getResources().getConfiguration().orientation
                == Configuration.ORIENTATION_PORTRAIT;

        int width = fullscreen
                ? dp(portrait ? 68 : 130)
                : dp(55);

        int height = fullscreen
                ? dp(portrait ? 42 : 80)
                : dp(34);

        FrameLayout.LayoutParams lp =
                new FrameLayout.LayoutParams(
                    width, height,
                    Gravity.TOP | Gravity.RIGHT
                );

        lp.setMargins(
                0,
                dp(portrait ? 8 : 12),
                dp(portrait ? 8 : 12),
                0
        );

        logo.setLayoutParams(lp);
        logo.setPadding(
                portrait ? 0 : dp(5),
                portrait ? 0 : dp(5),
                portrait ? 0 : dp(5),
                portrait ? 0 : dp(5)
        );

        logo.setBackgroundColor(
                portrait ? Color.TRANSPARENT : 0x880F0F0F
        );

        logo.bringToFront();
    }

    private void setBroadcastFullscreen(boolean fullscreen) {
        if (broadcastMiniBox == null) return;

        boolean wasFullscreen = broadcastFullscreen;

        if (fullscreen && !wasFullscreen) {
            broadcastPreviousOrientation = getRequestedOrientation();
        }

        broadcastFullscreen = fullscreen;

        if (broadcastPlayer != null) {
            broadcastPlayer.setVolume(1f);
        }

        FrameLayout.LayoutParams lp;

        if (fullscreen) {
            setRequestedOrientation(
                    android.content.pm.ActivityInfo.SCREEN_ORIENTATION_PORTRAIT
            );

            lp = new FrameLayout.LayoutParams(
                    FrameLayout.LayoutParams.MATCH_PARENT,
                    FrameLayout.LayoutParams.MATCH_PARENT,
                    Gravity.TOP | Gravity.LEFT
            );

            getWindow().getDecorView().setSystemUiVisibility(
                    View.SYSTEM_UI_FLAG_IMMERSIVE_STICKY
                    | View.SYSTEM_UI_FLAG_FULLSCREEN
                    | View.SYSTEM_UI_FLAG_HIDE_NAVIGATION
                    | View.SYSTEM_UI_FLAG_LAYOUT_FULLSCREEN
                    | View.SYSTEM_UI_FLAG_LAYOUT_HIDE_NAVIGATION
                    | View.SYSTEM_UI_FLAG_LAYOUT_STABLE
            );

            if (broadcastHideButton != null) {
                broadcastHideButton.setVisibility(View.GONE);
            }

        } else {
            if (wasFullscreen) {
                int restore = broadcastPreviousOrientation >= 0
                        ? broadcastPreviousOrientation
                        : android.content.pm.ActivityInfo.SCREEN_ORIENTATION_UNSPECIFIED;

                setRequestedOrientation(restore);
                broadcastPreviousOrientation = -1;
            }

            lp = new FrameLayout.LayoutParams(
                    broadcastMiniWidth(), broadcastMiniHeight(),
                    Gravity.TOP | Gravity.LEFT
            );
            lp.setMargins(dp(10), dp(68), 0, 0);

            getWindow().getDecorView().setSystemUiVisibility(0);

            if (broadcastHideButton != null) {
                broadcastHideButton.setVisibility(View.VISIBLE);
            }
        }

        broadcastMiniBox.setLayoutParams(lp);
        broadcastMiniBox.bringToFront();
        updateMiniLogoLayout(broadcastLogo, fullscreen);
    }

    private void stopBroadcastMiniPlayer(String id) {
        if (id != null && !id.isEmpty() &&
                !id.equals(activeBroadcastId)) {
            return;
        }

        if (broadcastFullscreen) {
            getWindow().getDecorView().setSystemUiVisibility(0);

            int restore = broadcastPreviousOrientation >= 0
                    ? broadcastPreviousOrientation
                    : android.content.pm.ActivityInfo.SCREEN_ORIENTATION_UNSPECIFIED;

            setRequestedOrientation(restore);
        }

        broadcastPreviousOrientation = -1;
        broadcastFullscreen = false;

        if (broadcastPlayerView != null) {
            broadcastPlayerView.setPlayer(null);
        }

        if (broadcastPlayer != null) {
            try {
                broadcastPlayer.pause();
                broadcastPlayer.stop();
                broadcastPlayer.release();
            } catch (Exception ignored) {}
        }

        if (broadcastMiniBox != null && root != null) {
            root.removeView(broadcastMiniBox);
        }

        broadcastPlayer = null;
        broadcastPlayerView = null;
        broadcastMiniBox = null;
        broadcastLogo = null;
        broadcastHideButton = null;
        activeBroadcastId = "";

        if (tvLogo != null && shouldShowTvLogo()) {
            tvLogo.setVisibility(View.VISIBLE);
        }
    }


    public class ISATVBridge {

        @JavascriptInterface
        public void showBroadcastFullscreen() {
            runOnUiThread(() -> setBroadcastFullscreen(true));
        }


        private final android.os.Handler broadcastMonitorHandler =
                new android.os.Handler(android.os.Looper.getMainLooper());

        private volatile String monitoredBroadcastId = "";

        private final Runnable broadcastMonitorRunnable = new Runnable() {
            @Override
            public void run() {
                final String id = monitoredBroadcastId;

                if (id == null || id.isEmpty()) {
                    return;
                }

                networkExecutor.execute(() -> {
                    boolean authoritative = false;
                    boolean found = false;

                    try {
                        String token = readMemberToken();

                        if (token != null && !token.isEmpty()) {
                            JSONObject response = memberHttp(
                                    "GET", "/member/broadcast", token, null
                            );

                            JSONArray broadcasts =
                                    response.optJSONArray("broadcasts");

                            authoritative =
                                    response.optInt("_http_status", 0) == 200
                                    && response.optBoolean("ok", false)
                                    && broadcasts != null;

                            if (authoritative) {
                                for (int i = 0; i < broadcasts.length(); i++) {
                                    JSONObject item = broadcasts.optJSONObject(i);

                                    if (item != null &&
                                            id.equals(item.optString("id", ""))) {
                                        found = true;
                                        break;
                                    }
                                }
                            }
                        }
                    } catch (Exception ignored) {
                        // Geçici ağ hatasında yayını kapatma; sonraki kontrolde dene.
                    }

                    if (authoritative && !found) {
                        runOnUiThread(() -> {
                            if (id.equals(monitoredBroadcastId)) {
                                monitoredBroadcastId = "";
                                broadcastMonitorHandler.removeCallbacks(
                                        broadcastMonitorRunnable
                                );
                                stopBroadcastMiniPlayer(id);
                            }
                        });
                    } else if (id.equals(monitoredBroadcastId)) {
                        broadcastMonitorHandler.postDelayed(
                                broadcastMonitorRunnable, 10000
                        );
                    }
                });
            }
        };

        private void startBroadcastMonitor(String id) {
            if (id == null || id.isEmpty()) return;

            broadcastMonitorHandler.removeCallbacks(
                    broadcastMonitorRunnable
            );

            monitoredBroadcastId = id;

            broadcastMonitorHandler.postDelayed(
                    broadcastMonitorRunnable, 10000
            );
        }

        private void cancelBroadcastMonitor(String id) {
            if (id == null || id.isEmpty() ||
                    id.equals(monitoredBroadcastId)) {
                monitoredBroadcastId = "";
                broadcastMonitorHandler.removeCallbacks(
                        broadcastMonitorRunnable
                );
            }
        }

try {

            runOnUiThread(() -> {
                int targetIndex = selectedIndex;

                if (url == null ||

                }

                try {
                    org.json.JSONArray queue = new org.json.JSONArray(

                    );

                        org.json.JSONObject one = new org.json.JSONObject();
                        one.put("name", name == null ? "İSAFLIX" : name);
                        one.put("url", url);
                        one.put("headers", new org.json.JSONObject(
                                headersJson == null ? "{}" : headersJson
                        ));
                        one.put("is_live", false);
                        queue.put(one);
                        targetIndex = 0;
                    }

                    java.io.File queueFile = new java.io.File(
                            getCacheDir(),
                            "isaflix-player-queue.json"
                    );

                    try (java.io.FileOutputStream output =
                                 new java.io.FileOutputStream(queueFile, false)) {
                        output.write(queue.toString().getBytes(
                                java.nio.charset.StandardCharsets.UTF_8
                        ));
                        output.flush();
                    }

                    Intent intent = new Intent(
                            MainActivity.this,
                            IsaflixPlayerActivity.class
                    );
                    intent.putExtra("stream_url", url);
                    intent.putExtra(
                            "channel_name",
                            name == null ? "İSAFLIX" : name
                    );
                    intent.putExtra(
                            "headers_json",
                            headersJson == null ? "{}" : headersJson
                    );
                    intent.putExtra("queue_file", queueFile.getAbsolutePath());
                    intent.putExtra("selected_index", targetIndex);

        @JavascriptInterface
        public void setFavorite(String channelJson, boolean favorite) {
            try {
                JSONObject channel = new JSONObject(channelJson);
                String url = channel.optString("url", "");
                if (url.isEmpty()) return;

                android.content.SharedPreferences prefs =
                        getSharedPreferences("isa_tv_data", MODE_PRIVATE);

                JSONArray old = new JSONArray(prefs.getString("favorites", "[]"));
                JSONArray updated = new JSONArray();

                for (int i = 0; i < old.length(); i++) {
                    JSONObject item = old.optJSONObject(i);
                    if (item == null) continue;
                    if (!url.equals(item.optString("url", ""))) {
                        updated.put(item);
                    }
                }

                if (favorite) updated.put(channel);

                prefs.edit().putString("favorites", updated.toString()).apply();
            } catch (Exception e) {
                Log.e(TAG, "Favori kaydedilemedi", e);
            }
        }



        // İSAFLİX kalıcı üye oturumu: token yalnızca AndroidKeyStore
        // AES/GCM ile şifrelenmiş olarak SharedPreferences içinde tutulur.
        private static final String MEMBER_API =
                "https://isa-admin-api.githuba85.workers.dev";
        private static final String MEMBER_PREFS =
                "isaflix_member_session";
        private static final String MEMBER_KEY_ALIAS =
                "isaflix_member_session_key";

        private android.content.SharedPreferences memberPrefs() {
            return getSharedPreferences(MEMBER_PREFS, MODE_PRIVATE);
        }

        private SecretKey memberSecretKey() throws Exception {
            KeyStore store = KeyStore.getInstance("AndroidKeyStore");
            store.load(null);

            java.security.Key existing = store.getKey(MEMBER_KEY_ALIAS, null);
            if (existing instanceof SecretKey) return (SecretKey) existing;

            KeyGenerator generator = KeyGenerator.getInstance(
                    KeyProperties.KEY_ALGORITHM_AES, "AndroidKeyStore"
            );
            generator.init(new KeyGenParameterSpec.Builder(
                    MEMBER_KEY_ALIAS,
                    KeyProperties.PURPOSE_ENCRYPT |
                            KeyProperties.PURPOSE_DECRYPT
            ).setBlockModes(KeyProperties.BLOCK_MODE_GCM)
             .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
             .setRandomizedEncryptionRequired(true)
             .build());

            return generator.generateKey();
        }

        private void saveMemberToken(String token) throws Exception {
            Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
            cipher.init(Cipher.ENCRYPT_MODE, memberSecretKey());

            byte[] encrypted = cipher.doFinal(
                    token.getBytes(StandardCharsets.UTF_8)
            );

            memberPrefs().edit()
                    .putString("token_cipher",
                            Base64.encodeToString(encrypted, Base64.NO_WRAP))
                    .putString("token_iv",
                            Base64.encodeToString(cipher.getIV(), Base64.NO_WRAP))
                    .apply();
        }

        private String readMemberToken() throws Exception {
            String encryptedText = memberPrefs().getString("token_cipher", "");
            String ivText = memberPrefs().getString("token_iv", "");
            if (encryptedText.isEmpty() || ivText.isEmpty()) return null;

            byte[] encrypted = Base64.decode(encryptedText, Base64.NO_WRAP);
            byte[] iv = Base64.decode(ivText, Base64.NO_WRAP);

            Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
            cipher.init(
                    Cipher.DECRYPT_MODE,
                    memberSecretKey(),
                    new GCMParameterSpec(128, iv)
            );

            return new String(
                    cipher.doFinal(encrypted),
                    StandardCharsets.UTF_8
            );
        }

        private String memberDeviceId() {
            String id = memberPrefs().getString("device_id", "");
            if (id.isEmpty()) {
                id = java.util.UUID.randomUUID().toString();
                memberPrefs().edit().putString("device_id", id).apply();
            }
            return id;
        }

        private JSONObject memberHttp(
                String method, String path, String token, JSONObject body
        ) throws Exception {
            HttpURLConnection connection = null;
            try {
                URL url = new URL(MEMBER_API + path);
                connection = (HttpURLConnection) url.openConnection();
                connection.setRequestMethod(method);
                connection.setConnectTimeout(15000);
                connection.setReadTimeout(20000);
                connection.setRequestProperty("Accept", "application/json");

                if (token != null && !token.isEmpty()) {
                    connection.setRequestProperty(
                            "Authorization", "Bearer " + token
                    );
                }

                if (body != null) {
                    connection.setDoOutput(true);
                    connection.setRequestProperty(
                            "Content-Type", "application/json; charset=utf-8"
                    );
                    byte[] bytes = body.toString().getBytes(StandardCharsets.UTF_8);
                    try (java.io.OutputStream output = connection.getOutputStream()) {
                        output.write(bytes);
                        output.flush();
                    }
                }

                int status = connection.getResponseCode();
                java.io.InputStream input = status >= 200 && status < 400
                        ? connection.getInputStream()
                        : connection.getErrorStream();

                String response = input == null ? "{}" : readStreamUtf8(input);
                JSONObject result;
                try {
                    result = new JSONObject(response);
                } catch (Exception ignored) {
                    result = new JSONObject();
                }
                result.put("_http_status", status);
                return result;
            } finally {
                if (connection != null) connection.disconnect();
            }
        }

                    );

                            response.optBoolean("ok", false) &&

                                .putString("member_json", member.toString())

                            result.put("error",
                                    "Üye kodu veya şifre hatalı.");

                    result = new JSONObject();

                        result.put("ok", false);
                        result.put("error",
                                "Bağlantı kurulamadı. Tekrar deneyin.");

        @JavascriptInterface
        public String readCatalogChunk(String offsetText, int requestedBytes) {
            try {
                long offset = Long.parseLong(offsetText);

                java.io.File file = new java.io.File(
                    getCacheDir(), "isaflix-web-katalog.json"
                );

                if (!file.isFile() || offset < 0 ||
                    offset >= file.length()) {
                    return "";
                }

                int size = Math.max(1024,
                    Math.min(65536, requestedBytes));

                int count = (int)Math.min(
                    (long) size, file.length() - offset
                );

                byte[] buffer = new byte[count];

                try (java.io.RandomAccessFile input =
                         new java.io.RandomAccessFile(file, "r")) {
                    input.seek(offset);
                    input.readFully(buffer);
                }

                return android.util.Base64.encodeToString(
                    buffer,
                    android.util.Base64.NO_WRAP
                );

            } catch (Exception e) {
                Log.e(TAG, "Catalog parça okuma hatası", e);
                return "";
            }
        }

        @JavascriptInterface
        public void fetchCatalogCatalog(String callbackId) {
            networkExecutor.execute(() -> {
                String error = null;
                boolean notModified = false;

                java.io.File cache = new java.io.File(
                    getCacheDir(), "isaflix-web-katalog.json"
                );

                android.content.SharedPreferences prefs =
                    getSharedPreferences("isaflix_web_catalog", MODE_PRIVATE);

                try {
                    URL treeUrl = new URL(
                        "https://api.github.com/repos/thechampisher/content-data/git/trees/main?recursive=1"
                    );

                    HttpURLConnection treeConnection =
                        (HttpURLConnection) treeUrl.openConnection();

                    treeConnection.setConnectTimeout(20000);
                    treeConnection.setReadTimeout(60000);
                    treeConnection.setRequestMethod("GET");
                    treeConnection.setRequestProperty(
                        "Accept", "application/vnd.github+json"
                    );
                    treeConnection.setRequestProperty(
                        "User-Agent", "Android-Catalog-Client"
                    );

                    String treeText;

                    try {
                        int status = treeConnection.getResponseCode();

                        if (status < 200 || status >= 300) {
                            throw new java.io.IOException(
                                "GitHub dosya listesi HTTP " + status
                            );
                        }

                        treeText = readStreamUtf8(treeConnection.getInputStream());
                    } finally {
                        treeConnection.disconnect();
                    }

                    org.json.JSONObject treeObject =
                        new org.json.JSONObject(treeText);

                    String treeSha = treeObject.optString("sha", "");

                    if (treeSha.isEmpty()) {
                        throw new java.io.IOException(
                            "GitHub depo sürümü alınamadı."
                        );
                    }

                    if (treeSha.equals(
                            prefs.getString("tree_sha", "")
                        ) && cache.isFile() && cache.length() > 0) {

                        new org.json.JSONObject(readUtf8File(cache));
                        notModified = true;

                    } else {
                        org.json.JSONArray entries =
                            treeObject.getJSONArray("tree");

                        java.util.ArrayList<String> jsonPaths =
                            new java.util.ArrayList<>();

                        for (int i = 0; i < entries.length(); i++) {
                            org.json.JSONObject entry =
                                entries.getJSONObject(i);

                            String path = entry.optString("path", "");

                            if (path.toLowerCase(java.util.Locale.ROOT)
                                    .endsWith(".json")
                                    && !path.startsWith(".git/")) {
                                jsonPaths.add(path);
                            }
                        }

                        java.util.Collections.sort(jsonPaths);

                        if (jsonPaths.isEmpty()) {
                            throw new java.io.IOException(
                                "Depoda JSON dosyası bulunamadı."
                            );
                        }

                        org.json.JSONObject combined =
                            new org.json.JSONObject();

                        int fileNumber = 0;

                        for (String path : jsonPaths) {
                            String encodedPath =
                                android.net.Uri.encode(path, "/");

                            URL fileUrl = new URL(
                                "https://raw.githubusercontent.com/thechampisher/content-data/main/"
                                + encodedPath
                            );

                            HttpURLConnection fileConnection =
                                (HttpURLConnection) fileUrl.openConnection();

                            fileConnection.setConnectTimeout(20000);
                            fileConnection.setReadTimeout(180000);
                            fileConnection.setRequestMethod("GET");
                            fileConnection.setRequestProperty(
                                "User-Agent", "Android-Catalog-Client"
                            );

                            String jsonText;

                            try {
                                int status = fileConnection.getResponseCode();

                                if (status < 200 || status >= 300) {
                                    throw new java.io.IOException(
                                        "JSON indirilemedi: " + path
                                        + " HTTP " + status
                                    );
                                }

                                jsonText = readStreamUtf8(
                                    fileConnection.getInputStream()
                                );
                            } finally {
                                fileConnection.disconnect();
                            }

                            Object parsed = new org.json.JSONTokener(
                                jsonText
                            ).nextValue();

                            if (parsed instanceof org.json.JSONObject) {
                                org.json.JSONObject object =
                                    (org.json.JSONObject) parsed;

                                java.util.Iterator<String> keys =
                                    object.keys();

                                while (keys.hasNext()) {
                                    String key = keys.next();
                                    Object value = object.get(key);

                                    if (!combined.has(key)) {
                                        combined.put(key, value);
                                    } else {
                                        Object previous = combined.get(key);

                                        if (previous instanceof org.json.JSONArray
                                                && value instanceof org.json.JSONArray) {
                                            org.json.JSONArray target =
                                                (org.json.JSONArray) previous;
                                            org.json.JSONArray incoming =
                                                (org.json.JSONArray) value;

                                            for (int j = 0; j < incoming.length(); j++) {
                                                target.put(incoming.get(j));
                                            }
                                        } else if (previous instanceof org.json.JSONObject
                                                && value instanceof org.json.JSONObject) {
                                            org.json.JSONObject target =
                                                (org.json.JSONObject) previous;
                                            org.json.JSONObject incoming =
                                                (org.json.JSONObject) value;

                                            java.util.Iterator<String> nested =
                                                incoming.keys();

                                            while (nested.hasNext()) {
                                                String nestedKey = nested.next();
                                                target.put(
                                                    nestedKey,
                                                    incoming.get(nestedKey)
                                                );
                                            }
                                        } else {
                                            combined.put(key, value);
                                        }
                                    }
                                }

                            } else {
                                String name = path.substring(
                                    path.lastIndexOf('/') + 1
                                ).replaceFirst("(?i)\\.json$", "");

                                combined.put(name, parsed);
                            }

                            fileNumber++;
                            reportCatalogProgress(
                                callbackId,
                                "download",
                                Math.min(
                                    70,
                                    fileNumber * 70 / jsonPaths.size()
                                )
                            );
                        }

                        String finalJson = combined.toString();

                        new org.json.JSONObject(finalJson);

                        java.io.File temp = new java.io.File(
                            getCacheDir(), "catalog-all-json.tmp"
                        );

                        try (
                            java.io.FileOutputStream output =
                                new java.io.FileOutputStream(temp, false)
                        ) {
                            output.write(
                                finalJson.getBytes(
                                    java.nio.charset.StandardCharsets.UTF_8
                                )
                            );
                            output.getFD().sync();
                        }

                        if (!temp.renameTo(cache)) {
                            if (cache.exists() && !cache.delete()) {
                                throw new java.io.IOException(
                                    "Eski katalog önbelleği değiştirilemedi."
                                );
                            }

                            if (!temp.renameTo(cache)) {
                                throw new java.io.IOException(
                                    "Yeni katalog önbelleğe taşınamadı."
                                );
                            }
                        }

                        prefs.edit()
                            .putString("tree_sha", treeSha)
                            .remove("etag")
                            .remove("modified")
                            .apply();
                    }

                } catch (Exception ex) {
                    error = ex.getMessage() == null
                        ? "JSON kataloğu alınamadı."
                        : ex.getMessage();

                    Log.e(TAG, "JSON depo güncelleme hatası", ex);
                }

                final String failure = error;
                final boolean unchanged = notModified;
                final String id = callbackId == null
                    ? "0" : callbackId.replaceAll("[^0-9]", "");

                final long catalogSize =
                    failure == null && cache.isFile()
                    ? cache.length() : 0L;

                runOnUiThread(() -> {
                    if (webView == null) return;

                    webView.evaluateJavascript(
                        "window.__isaCatalogReady(" + id + "," +
                        catalogSize + "," +
                        JSONObject.quote(failure) + "," +
                        (unchanged ? "true" : "false") + ");",
                        null
                    );
                });
            });
        }

        private String readStreamUtf8(java.io.InputStream input)
                throws Exception {
            StringBuilder result = new StringBuilder();

            try (
                java.io.BufferedReader reader =
                    new java.io.BufferedReader(
                        new java.io.InputStreamReader(
                            input,
                            java.nio.charset.StandardCharsets.UTF_8
                        ),
                        65536
                    )
            ) {
                char[] buffer = new char[32768];
                int count;

                while ((count = reader.read(buffer)) != -1) {
                    result.append(buffer, 0, count);
                }
            }

            return result.toString();
        }

        private String readUtf8File(java.io.File file) throws Exception {
            StringBuilder result = new StringBuilder();
            try (
                java.io.BufferedReader reader = new java.io.BufferedReader(
                    new java.io.InputStreamReader(
                        new java.io.FileInputStream(file),
                        StandardCharsets.UTF_8
                    ), 65536
                )
            ) {
                char[] buffer = new char[32768];
                int n;
                while ((n = reader.read(buffer)) != -1) {
                    result.append(buffer, 0, n);
                }
            }
            return result.toString();
        }

        private void decryptCatalogFile(
            java.io.File source,
            java.io.File destination,
            String callbackId
        ) throws Exception {

            try (
                java.io.BufferedInputStream input =
                    new java.io.BufferedInputStream(
                        new java.io.FileInputStream(source), 65536
                    )
            ) {
                byte[] header = new byte[4];
                int count = 0;
                while (count < 4) {
                    int n = input.read(header, count, 4-count);
                    if (n < 0) throw new java.io.IOException("Dosya eksik.");
                    count += n;
                }

                boolean encrypted =
                    header[0] == 68 && header[1] == 84 &&
                    header[2] == 86 && header[3] == 49;

                java.io.InputStream plain = input;

                if (encrypted) {
                    byte[] iv = new byte[12];
                    int filled = 0;

                    while (filled < 12) {
                        int n = input.read(iv, filled, 12-filled);
                        if (n < 0)
                            throw new java.io.IOException("DTV1 IV eksik.");
                        filled += n;
                    }

                    byte[] digest = java.security.MessageDigest
                        .getInstance("SHA-256")
                        .digest(
                            (new String(new char[]{100,111,109,105,110,111}) + "|tv-cat|alog-v1|-2026")
                            .getBytes(StandardCharsets.UTF_8)
                        );

                    javax.crypto.Cipher cipher =
                        javax.crypto.Cipher.getInstance("AES/GCM/NoPadding");

                    cipher.init(
                        javax.crypto.Cipher.DECRYPT_MODE,
                        new javax.crypto.spec.SecretKeySpec(digest, "AES"),
                        new javax.crypto.spec.GCMParameterSpec(128, iv)
                    );

                    final long cipherTotal = Math.max(1L, source.length() - 16L);

                    plain = new javax.crypto.CipherInputStream(
                        new java.io.FilterInputStream(input) {
                            private long consumed = 0;
                            private int lastPercent = -1;
                            private long lastReport = 0;

                            private void updateProgress(long amount) {
                                consumed += amount;

                                int percent = (int)Math.min(
                                    90L, 70L + consumed * 20L / cipherTotal
                                );

                                long now = android.os.SystemClock.elapsedRealtime();

                                if (percent != lastPercent &&
                                    (now - lastReport >= 150 || percent == 99)) {
                                    reportCatalogProgress(
                                        callbackId, "load", percent
                                    );
                                    lastPercent = percent;
                                    lastReport = now;
                                }
                            }

                            @Override
                            public int read(byte[] b, int off, int len)
                                throws java.io.IOException {
                                int n = super.read(b, off, len);
                                if (n > 0) updateProgress(n);
                                return n;
                            }

                            @Override
                            public int read() throws java.io.IOException {
                                int value = super.read();
                                if (value != -1) updateProgress(1);
                                return value;
                            }
                        },
                        cipher
                    );
                } else {
                    java.io.PushbackInputStream push =
                        new java.io.PushbackInputStream(input, 4);
                    push.unread(header);
                    plain = push;
                }

                java.io.File temp = new java.io.File(
                    destination.getParentFile(),
                    destination.getName() + ".tmp"
                );

                try (
                    java.io.InputStream decoded = plain;
                    java.io.BufferedOutputStream output =
                        new java.io.BufferedOutputStream(
                            new java.io.FileOutputStream(temp), 65536
                        )
                ) {
                    byte[] buffer = new byte[65536];
                    int n;

                    while ((n = decoded.read(buffer)) != -1) {
                        output.write(buffer, 0, n);
                    }

                    output.flush();
                }

                reportCatalogProgress(callbackId, "load", 90);

                if (!temp.renameTo(destination)) {
                    try (
                        java.io.InputStream in = new java.io.FileInputStream(temp);
                        java.io.OutputStream out =
                            new java.io.FileOutputStream(destination)
                    ) {
                        byte[] buffer = new byte[65536];
                        int n;
                        while ((n = in.read(buffer)) != -1)
                            out.write(buffer, 0, n);
                    }
                    temp.delete();
                }
            }
        }

        @JavascriptInterface
public void fetchText(String address, String callbackId) {
    networkExecutor.execute(() -> {
        String result = null;
        String error = null;

        for (int attempt = 1; attempt <= 3; attempt++) {
            HttpURLConnection connection = null;

            try {
                URL url = new URL(address);

                if (!"http".equalsIgnoreCase(url.getProtocol())
                        && !"https".equalsIgnoreCase(url.getProtocol())) {
                    throw new Exception("HTTP/HTTPS adresi gerekli.");
                }

                connection = (HttpURLConnection) url.openConnection();
                connection.setConnectTimeout(20000);
                connection.setReadTimeout(120000);
                connection.setRequestMethod("GET");
                connection.setInstanceFollowRedirects(true);
                connection.setRequestProperty("Accept", "*/*");
                connection.setRequestProperty("Accept-Encoding", "identity");
                connection.setRequestProperty(
                    "User-Agent",
                    "Mozilla/5.0 (Linux; Android 11) AppleWebKit/537.36 "
                    + "(KHTML, like Gecko) Chrome/120.0 Mobile Safari/537.36"
                );

                int code = connection.getResponseCode();

                if (code < 200 || code >= 300) {
                    throw new java.io.IOException("HTTP " + code);
                }

                StringBuilder output = new StringBuilder();

                try (BufferedReader reader = new BufferedReader(
                        new InputStreamReader(
                            connection.getInputStream(),
                            StandardCharsets.UTF_8))) {

                    String line;
                    while ((line = reader.readLine()) != null) {
                        output.append(line).append('\n');

                    if (output.length() > 64 * 1024 * 1024) {
                            throw new java.io.IOException("Kaynak 64 MB sınırını aştı.");
                        }
}
                }

                result = output.toString();
                error = null;
                break;

            } catch (Exception e) {
                error = e.getMessage();
                Log.w(TAG, "Kaynak indirme denemesi " + attempt + "/3", e);

                if (attempt < 3) {
                    try {
                        Thread.sleep(attempt * 1200L);
                    } catch (InterruptedException interrupted) {
                        Thread.currentThread().interrupt();
                        error = "İşlem iptal edildi.";
                        break;
                    }
                }
            } finally {
                if (connection != null) {
                    connection.disconnect();
                }
            }
        }

        final String response = result;
        final String failure = error;
        final String safeId = callbackId == null
                ? "0"
                : callbackId.replaceAll("[^0-9]", "");

        runOnUiThread(() -> {
            if (webView == null) return;

            String js = "window.__isaFetchDone("
                    + safeId + ","
                    + JSONObject.quote(response) + ","
                    + JSONObject.quote(failure) + ");";

            webView.evaluateJavascript(js, null);
        });
    });

        @JavascriptInterface
        public void fetchTextWithHeaders(String address, String headersJson, String callbackId) {
            networkExecutor.execute(() -> {
                String result = null;
                String error = null;
                HttpURLConnection connection = null;
                try {
                    URL url = new URL(address);
                    if (!"http".equalsIgnoreCase(url.getProtocol()) &&
                            !"https".equalsIgnoreCase(url.getProtocol())) {
                        throw new Exception("HTTP/HTTPS adresi gerekli.");
                    }
                    connection = (HttpURLConnection) url.openConnection();
                    connection.setConnectTimeout(20000);
                    connection.setReadTimeout(120000);
                    connection.setRequestMethod("GET");
                    connection.setInstanceFollowRedirects(true);
                    connection.setRequestProperty("Accept", "*/*");
                    connection.setRequestProperty("Accept-Encoding", "identity");

                    JSONObject supplied = new JSONObject(headersJson == null ? "{}" : headersJson);
                    java.util.Iterator<String> keys = supplied.keys();
                    while (keys.hasNext()) {
                        String key = keys.next();
                        String value = supplied.optString(key, "");
                        if (key.matches("(?i)[a-z0-9-]+") &&
                                !key.matches("(?i)host|content-length|connection|transfer-encoding|accept-encoding") &&
                                !value.contains("\r") && !value.contains("\n")) {
                            connection.setRequestProperty(key, value);
                        }
                    }

                    int code = connection.getResponseCode();
                    if (code < 200 || code >= 300) throw new java.io.IOException("HTTP " + code);

                    StringBuilder output = new StringBuilder();
                    try (BufferedReader reader = new BufferedReader(
                            new InputStreamReader(connection.getInputStream(), StandardCharsets.UTF_8))) {
                        String line;
                        while ((line = reader.readLine()) != null) {
                            output.append(line).append('\n');
                            if (output.length() > 64 * 1024 * 1024) {
                                throw new Exception("Kaynak 64 MB sınırını aştı.");
                            }
                        }
                    }
                    result = output.toString();
                } catch (Exception e) {
                    error = e.getMessage() == null ? "İstek başarısız." : e.getMessage();
                    Log.w(TAG, "Başlıklı kaynak isteği başarısız", e);
                } finally {
                    if (connection != null) connection.disconnect();
                }

                final String response = result;
                final String failure = error;
                final String safeId = callbackId == null ? "0" : callbackId.replaceAll("[^0-9]", "");
                runOnUiThread(() -> {
                    if (webView == null) return;
                    String callback = "window.__isaFetchDone(" + safeId + "," +
                            JSONObject.quote(response) + "," +
                            JSONObject.quote(failure) + ");";
                    webView.evaluateJavascript(callback, null);
                });
            });
        }

@JavascriptInterface
        public void cachePut(String key, String value) {
            try {
                java.io.File directory = new java.io.File(
                        getFilesDir(),
                        "isa-source-cache"
                );

                if (!directory.exists()) {
                    directory.mkdirs();
                }

                String fileName = Integer.toHexString(
                        key == null ? 0 : key.hashCode()
                ) + ".cache";

                java.io.File file = new java.io.File(
                        directory,
                        fileName
                );

                try (java.io.FileOutputStream output =
                             new java.io.FileOutputStream(file, false)) {
                    output.write(
                            (value == null ? "" : value)
                                    .getBytes(java.nio.charset.StandardCharsets.UTF_8)
                    );
                    output.flush();
                }

            } catch (Exception e) {
                android.util.Log.e("ISATV", "Önbellek yazma hatası", e);
            }
        }

        @JavascriptInterface
        public void cacheGet(String key, String callbackId) {
            String result = null;
            String error = null;

            try {
                java.io.File directory = new java.io.File(
                        getFilesDir(),
                        "isa-source-cache"
                );

                java.io.File file = new java.io.File(
                        directory,
                        Integer.toHexString(
                                key == null ? 0 : key.hashCode()
                        ) + ".cache"
                );

                if (file.isFile()) {
                    StringBuilder content = new StringBuilder();

                    try (java.io.FileInputStream input =
                                 new java.io.FileInputStream(file)) {
                        byte[] buffer = new byte[8192];
                        int count;

                        while ((count = input.read(buffer)) != -1) {
                            content.append(
                                    new String(
                                            buffer,
                                            0,
                                            count,
                                            java.nio.charset.StandardCharsets.UTF_8
                                    )
                            );
                        }
                    }

                    result = content.toString();
                }

            } catch (Exception e) {
                error = e.getMessage();
            }

            final String finalResult = result;
            final String finalError = error;

            final String safeId = callbackId == null
                    ? "0"
                    : callbackId.replaceAll("[^0-9]", "");

            runOnUiThread(() -> {
                if (webView == null) return;

                String valueJson = finalResult == null
                        ? "null"
                        : org.json.JSONObject.quote(finalResult);

                String errorJson = finalError == null
                        ? "null"
                        : org.json.JSONObject.quote(finalError);

                webView.evaluateJavascript(
                        "window.__isaCacheDone(" +
                                safeId + "," +
                                valueJson + "," +
                                errorJson + ");",
                        null
                );
            });
        }

    }

    @Override
    protected void onActivityResult(
            int requestCode,
            int resultCode,
            Intent data) {

        super.onActivityResult(requestCode, resultCode, data);

        if (requestCode == FILE_REQUEST && fileCallback != null) {
            Uri[] result =
                    WebChromeClient.FileChooserParams.parseResult(
                            resultCode,
                            data
                    );

            fileCallback.onReceiveValue(result);
            fileCallback = null;
        }
    }

    @Override
    protected void onResume() {
        super.onResume();
        if (webView != null) {
            webView.postDelayed(() -> {
                if (webView != null) {
                    webView.evaluateJavascript(
                        "if(window.homepage && typeof window.homepage.restoreLastPlayedChannel === 'function'){" +
                        "window.homepage.restoreLastPlayedChannel();" +
                        "if (window.isaBroadcastPoll) window.isaBroadcastPoll();}",
                        null
                    );
                }
            }, 250);
        }
    }

    @Override
    public void onBackPressed() {
        if (broadcastFullscreen) {
            setBroadcastFullscreen(false);
            return;
        }
        if (webView == null) {
            handleExitBack();
            return;
        }

        webView.evaluateJavascript(
                "(function(){return window.isaHandleBack ? window.isaHandleBack() : false;})()",
                value -> {
                    if ("true".equals(value)) {
                        backPressedAt = 0L;
                    } else {
                        handleExitBack();
                    }
                }
        );
    }

    private void handleExitBack() {
        long now = System.currentTimeMillis();
        if (now - backPressedAt < 2000L) {
            finish();
            return;
        }

        backPressedAt = now;
        Toast.makeText(
                this,
                "Çıkmak için geri tuşuna tekrar basın",
                Toast.LENGTH_SHORT
        ).show();
    }

    @Override
    protected void onDestroy() {
        if (fileCallback != null) {
            fileCallback.onReceiveValue(null);
            fileCallback = null;
        }

        stopBroadcastMiniPlayer(null);
        networkExecutor.shutdownNow();

        if (webView != null) {
            webView.destroy();
            webView = null;
        }

        super.onDestroy();
    }
